using System.Text.Json;using System.Text.Json.Serialization;
public record Noul(string Instructions);
public record Choice(string Instructions, string[] Options);
public union Question(Noul, Choice);
public static class P { public static void Main(){
 var t=typeof(Question); Console.WriteLine($"{t.IsValueType} {t.BaseType} ifaces={string.Join(",",t.GetInterfaces().Select(i=>i.Name))}");
 foreach(var p in t.GetProperties()) Console.WriteLine("prop "+p.Name+":"+p.PropertyType);
 Question q = new Noul("x"); Console.WriteLine(JsonSerializer.Serialize(q));
 try{ var r=JsonSerializer.Deserialize<Question>("{\"type\":\"noul\",\"Instructions\":\"x\"}"); Console.WriteLine("deser "+r);}catch(Exception e){Console.WriteLine("deser fail: "+e.GetType().Name+": "+e.Message.Split('.')[0]);}
}}
